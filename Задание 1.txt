
function hasTwoCubeSums(n) {
  let wc = 0;
  const limit = Math.cbrt(n / 2);

  for (let a = 1; a <= limit; a++) {
    const a3 = a * a * a;
    const b3 = n - a3;
    const b = Math.round(Math.cbrt(b3));

    if (b > 0 && b * b * b === b3) {
      wc++; 
      if (wc >= 2) {
        return true;
      }
    }
  } 

  return false; 
}
